function _cosine29(a, b) {
  // Specialized for 29-D — branchless tight loop, no length checks
  let dot = 0, na = 0, nb = 0;
  for (let i = 0; i < 29; i++) {
    const x = a[i] || 0, y = b[i] || 0;
    dot += x * y; na += x * x; nb += y * y;
  }
  if (na === 0 || nb === 0) return 0;
  return dot / (Math.sqrt(na) * Math.sqrt(nb));
}
